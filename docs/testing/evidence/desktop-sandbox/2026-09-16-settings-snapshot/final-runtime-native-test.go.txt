//go:build darwin

// INPUT: 已加载的权限设置与其后发生的内容失效。
// OUTPUT: 动态权限更新不得因重新读取未知设置而删除原有 deny。
// POS: 普通配置快照的原生运行期回归。
package runtime

import (
	"context"
	"os"
	"path/filepath"
	"strings"
	"testing"

	tools "github.com/nexus-research-lab/nexus-agent-sdk-go/internal/tool/executor"
	"github.com/nexus-research-lab/nexus-agent-sdk-go/permission"
	"github.com/nexus-research-lab/nexus-agent-sdk-go/protocol"
)

// TestDarwinSettingsIntegrity 在内存权限发生改变前拒绝失效的源文件。
func TestDarwinSettingsIntegrity(t *testing.T) {
	for _, change := range []string{"malformed", "deleted", "relaxed"} {
		t.Run(change, func(t *testing.T) {
			cfg := contextFileFixture(t)
			path := filepath.Join(cfg.ConfigDir, "settings.json")
			const original = `{"permissions":{"deny":["Bash"]}}`
			writeContextFixture(t, path, original)
			rt, err := New(cfg)
			if err != nil {
				t.Fatal(err)
			}
			switch change {
			case "malformed":
				writeContextFixture(t, path, `{"permissions":`)
			case "deleted":
				if err := os.Remove(path); err != nil {
					t.Fatal(err)
				}
			case "relaxed":
				writeContextFixture(t, path, `{}`)
			}
			err = rt.ApplyPermissionUpdates(context.Background(), []permission.Update{{
				Type: "setMode", Mode: permission.ModeBypassPermissions, Destination: permission.UpdateDestinationSession,
			}})
			if err == nil {
				t.Fatal("changed settings admitted permission update and lost mandatory rules")
			}
			assertBlocked := func(err error) {
				t.Helper()
				if err == nil || !strings.Contains(err.Error(), "settings") {
					t.Fatalf("missing settings rejection: %v", err)
				}
			}
			emit := func(protocol.ReceivedMessage) error { t.Fatal("unknown settings reached model emitter"); return nil }
			_, err = rt.RunQuery(t.Context(), QueryConfig{}, emit)
			assertBlocked(err)
			assertBlocked(rt.CompactSessionNow(t.Context(), QueryConfig{}, "", emit))
			_, err = rt.tools.Run(t.Context(), tools.Request{Name: "Bash", Input: map[string]any{"command": "echo unsafe"}}, tools.PermissionContext{Mode: "bypassPermissions"})
			assertBlocked(err)
			writeContextFixture(t, path, original)
			if err := rt.checkConfiguration(t.Context()); err != nil {
				t.Fatal(err)
			}
			result, err := rt.tools.Run(t.Context(), tools.Request{Name: "Bash", Input: map[string]any{"command": "echo unsafe"}}, tools.PermissionContext{Mode: "bypassPermissions"})
			if err == nil && !result.Result.IsError {
				t.Fatal("restored configuration lost original deny")
			}
		})
	}
}
