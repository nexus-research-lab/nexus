import datetime
import hashlib
import json
import shutil
from pathlib import Path

scratch = Path(__file__).resolve().parent
nexus = Path('/Users/berhand/program/Work/Nexus/worktrees/nexus/desktop-sandbox')
sdk = Path('/private/tmp/nexus-sandbox-capabilities.60SUhq/sdk')
destination = nexus / 'docs/testing/evidence/desktop-sandbox/2026-09-16-settings-snapshot'
baseline_dir = Path((scratch / 'baseline-driver.jsonl').read_text().splitlines()[0].split(': ', 1)[1])
baseline = json.loads((baseline_dir / 'report.json').read_text())
assert baseline['passed'] and not baseline['releaseAccepted'] and not baseline['sdkWorktreeChanges']
names = ['baseline-driver', 'sdk-race-final', 'sdk-controls', 'sdk-vet', 'bridge-race', 'bridge-vet', 'native-recovery-final',
         'nexus-race', 'nexus-vet', 'architecture', 'evidence-parser',
         'sdk-windows', 'sdk-linux', 'bridge-windows', 'bridge-linux', 'bridge-pinned-real-process']
records = {}
for name in names:
    record = json.loads((scratch / (name + '.result.json')).read_text())
    assert record['exitCode'] == 0, (name, record)
    record['log'] = name + '.jsonl'
    events = []
    for line in (scratch / record['log']).read_text().splitlines():
        try:
            event = json.loads(line)
        except ValueError:
            continue
        if isinstance(event, dict):
            events.append(event)
    if events:
        failures = [e for e in events if e.get('Action') in ('fail', 'build-fail')]
        assert not failures, (name, failures)
        record['topLevelPassed'] = sum(e.get('Action') == 'pass' and 'Test' in e and '/' not in e['Test'] for e in events)
        record['subtestsPassed'] = sum(e.get('Action') == 'pass' and '/' in e.get('Test', '') for e in events)
        record['skipped'] = [{'package': e.get('Package'), 'test': e.get('Test')} for e in events if e.get('Action') == 'skip']
    records[name] = record
matrix = records['bridge-pinned-real-process']
assert matrix['topLevelPassed'] == 10 and matrix['subtestsPassed'] == 18 and not matrix['skipped'], matrix
required = [t for c in baseline['checks'] for t in c.get('passedTests', [])]
assert all(c['exitCode'] == 0 for c in baseline['checks'])
assert not destination.exists(), destination
destination.mkdir(parents=True)

    shutil.copyfile(source, destination / source.name)

for source in initial_dir.glob('*context-recovery*.log'):
    shutil.copyfile(source, destination / ('initial-' + source.name))
for name in names:

        shutil.copyfile(scratch / (name + suffix), destination / (name + suffix))
for source, target in [('before.jsonl', 'reproduction-before.jsonl'), ('before.result.json', 'reproduction-before.result.json'),

                       ('matrix.py', 'run-matrix.py.txt'), ('archive.py', 'archive.py.txt'),
                       ('baseline-driver-initial.jsonl', 'baseline-driver-initial.jsonl'),

                       ('internal/agent/runtime/settings_integrity_darwin_test.go', 'final-runtime-native-test.go.txt'),
                       ('cmd/nxs/settings_integrity_test.go', 'final-control-test.go.txt')]:
    shutil.copyfile(sdk / source, destination / target)

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

sources = ['go.mod', 'go.sum', 'internal/runtime/process_policy.go', 'internal/runtime/process_policy_test.go',
           'internal/runtime/clientopts/desktop_sandbox.go', 'internal/runtime/clientopts/desktop_sandbox_test.go',
           'internal/runtime/clientopts/desktop_sandbox_integration_test.go', 'scripts/desktop/check-sandbox-baseline.mjs']
legacy = []
for variable, value in matrix['environment'].items():
    if 'LEGACY' in variable:
        binary = Path(value)
        entry = {'variable': variable, 'path': value, 'sha256': sha(binary)}
        report_path = binary.parent / 'report.json'
        if report_path.exists():
            entry['sdkRevision'] = json.loads(report_path.read_text()).get('sdkRevision')
        legacy.append(entry)

    ]
}
